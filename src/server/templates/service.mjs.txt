export default ({ service, fmt }) => service({
  type: '__NAME__',
  name: '__TITLE__',
  icon: 'pulse',
  auth: 'bearer',                    // or 'none', 'token', 'basic', { header: 'X-Api-Key' }, { query: 'apikey' }, { login: '/api/login' }
  get: { status: '/api/status' },    // each one is fetched, and its answer arrives as d.status
  stats: d => [
    { label: 'Used', value: fmt.bytes(d.status.used) },
    { label: 'Items', value: d.status.items },
  ],
  sample: { status: { used: 1610612736, items: 1204 } },
  expect: { Used: fmt.bytes(1610612736), Items: '1,204' },
});
